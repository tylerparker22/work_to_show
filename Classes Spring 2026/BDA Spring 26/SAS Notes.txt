/* ANOVA */
/* LOAD FILE */
proc import 
    datafile="/home/u64442749/sasuser.v94/new micro treatment.csv"
    out=apc
    dbms=csv
    replace;
    guessingrows=max;
run;

/* ANOVA model */
proc glm data=apc;
class trt;
model log10APC = trt;
run;
quit;

/* ANOVA + Multiple comparisons */
proc glm data=apc;
class trt;
model log10APC = trt;
means trt / duncan tukey hovtest=levene; 
run;
quit;
